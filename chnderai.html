    // ----- 4. आवाज़ें लोड करना (सबसे जरूरी) -----

    function loadVoices() {

        const voices = window.speechSynthesis.getVoices();

        if (!voices || voices.length === 0) {

            setTimeout(loadVoices, 200);

            return;

        }

        voiceSelect.innerHTML = '<option value="">आवाज़ चुनें...</option>';

        voices.forEach(voice => {

            const opt = document.createElement('option');

            opt.value = voice.name;

            opt.textContent = `${voice.name} (${voice.lang})`;

            voiceSelect.appendChild(opt);

        });

        // हिंदी या अंग्रेज़ी को डिफॉल्ट सेट करें

        const defaultVoice = voices.find(v => v.lang === 'hi-IN') || voices.find(v => v.lang.startsWith('en')) || voices[0];

        if (defaultVoice) {

            voiceSelect.value = defaultVoice.name;

            currentVoice = defaultVoice;

        }

    }

    if (window.speechSynthesis) {

        window.speechSynthesis.onvoiceschanged = loadVoices;

        setTimeout(loadVoices, 300);

    }

    voiceSelect.addEventListener('change', function() {

        const voices = window.speechSynthesis.getVoices();

        currentVoice = voices.find(v => v.name === this.value) || null;

    });


    // ----- 5. आवाज़ में बोलना (स्पीकर फंक्शन) -----

    function speakText(btn, text) {

        if (!window.speechSynthesis) return;

        window.speechSynthesis.cancel();

        const utterance = new SpeechSynthesisUtterance(text);

        if (currentVoice) utterance.voice = currentVoice;

        else {

            const voices = window.speechSynthesis.getVoices();

            utterance.voice = voices.find(v => v.lang === 'hi-IN') || voices.find(v => v.lang.startsWith('en')) || null;

        }

        utterance.rate = 0.9; utterance.pitch = 1; utterance.volume = 1;

        window.speechSynthesis.speak(utterance);

        if (btn) {

            const icon = btn.querySelector('i');

            if (icon) {

                icon.className = 'fas fa-spinner fa-pulse';

                setTimeout(() => { icon.className = 'fas fa-volume-up'; }, 2000);

            }

        }

    }


    // ----- 6. मैसेज एड करना और भाषा डिटेक्ट करना -----

    function detectLanguage(text) {

        if (/[\u0900-\u097F]/.test(text)) return '🇮🇳 हिंदी';

        if (/[\u0B80-\u0BFF]/.test(text)) return '🇮🇳 तमिल';

        if (/[\u0C00-\u0C7F]/.test(text)) return '🇮🇳 तेलुगु';

        if (/[\u0A80-\u0AFF]/.test(text)) return '🇮🇳 गुजराती';

        if (/[\u0980-\u09FF]/.test(text)) return '🇮🇳 बांग्ला';

        return '🌐 अंग्रेज़ी';

    }


    function addMessage(text, sender) {

        const div = document.createElement('div');

        div.className = `msg ${sender === 'user' ? 'user' : 'bot'}`;

        if (sender === 'user') {

            div.textContent = text;

        } else {

            const lang = detectLanguage(text);

            const safeText = text.replace(/'/g, "\\'").replace(/"/g, '\\"');

            div.innerHTML = `

                <i class="fas fa-robot" style="color:#ffd700; margin-right:6px;"></i> 

                ${text}

                <span class="lang-tag">${lang}</span>

                <button class="speaker" onclick="speakText(this, '${safeText}')">

                    <i class="fas fa-volume-up"></i>

                </button>

            `;

            if (autoSpeak.checked) {

                setTimeout(() => {

                    const spk = div.querySelector('.speaker');

                    if (spk) speakText(spk, text);

                }, 300);

            }

        }

        chatMessages.insertBefore(div, typingDots);

        chatMessages.scrollTop = chatMessages.scrollHeight;

    }


    // ----- 7. AI से जवाब लेना (Gemini API) -----

    async function getAIResponse(userText) {

        chatHistory.push({ role: "user", parts: [{ text: userText }] });

        const systemInstruction = `तुम एक बुद्धिमान भारतीय AI असिस्टेंट हो। नाम: 'Mega India AI'।

        नियम: हमेशा उसी भाषा में जवाब दो जिसमें यूजर ने पूछा है। पढ़ाई, कानून, डॉक्टर, खेती, मंडी, AI, फिल्म, सब में मदद करो।`;

        const body = {

            contents: chatHistory,

            systemInstruction: { parts: [{ text: systemInstruction }] },

            generationConfig: { temperature: 0.7, maxOutputTokens: 900 }

        };

        const res = await fetch(API_URL, {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify(body)

        });

        if (!res.ok) {

            const errText = await res.text();

            throw new Error(`API Error (${res.status}): ${errText}`);

        }

        const data = await res.json();

        const reply = data.candidates?.[0]?.content?.parts?.[0]?.text || "क्षमा करें, जवाब नहीं मिला।";

        chatHistory.push({ role: "model", parts: [{ text: reply }] });

        if (chatHistory.length > 20) chatHistory = chatHistory.slice(-20);

        return reply;

    }


    // ----- 8. मैसेज भेजना (Send) -----

    async function sendChatMessage() {

        const text = chatInput.value.trim();

        if (!text) return;

        if (API_KEY === "YOUR_API_KEY") {

            addMessage('⚠️ कृपया पहले कोड में "YOUR_API_KEY" की जगह अपनी असली Google API Key डालें! (ऊपर JS में)', 'bot');

            chatInput.value = '';

            return;

        }

        addMessage(text, 'user');

        chatInput.value = '';

        sendBtn.disabled = true;

        typingDots.style.display = 'flex';

        chatMessages.scrollTop = chatMessages.scrollHeight;


        try {

            const reply = await getAIResponse(text);

            typingDots.style.display = 'none';

            addMessage(reply, 'bot');

        } catch (err) {

            typingDots.style.display = 'none';

            addMessage(`❌ त्रुटि: ${err.message}. कृपया API Key चेक करें या इंटरनेट कनेक्शन देखें।`, 'bot');

            console.error(err);

        }

        sendBtn.disabled = false;

        chatMessages.scrollTop = chatMessages.scrollHeight;

    }

    function switchTab(tab) {

        const homePage = document.getElementById('homePage');

        const chatWin = document.getElementById('chatWindow');

        const navHome = document.getElementById('navHome');

        const navChat = document.getElementById('navChat');

        if (tab === 'home') {

            homePage.style.display = 'block';

            chatWin.classList.remove('active');

            chatWin.style.display = 'none';

            navHome.classList.add('active');

            navChat.classList.remove('active');

        } else {

            homePage.style.display = 'none';

            chatWin.style.display = 'flex';

            chatWin.classList.add('active');

            navChat.classList.add('active');

            navHome.classList.remove('active');

            // चैट इनपुट पर फोकस करें

            setTimeout(() => chatInput.focus(), 100);

        }

    }
